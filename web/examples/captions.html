A WebVTT track drawn over the 3D scene by the renderer's overlay pass — the same code path cargo run --example captions_render uses natively. The text is rasterized only when the active cue changes, so the per-frame cost is one texture upload. Nothing here needs a font file: the built-in bitmap face ships in the wasm.

The whole integration

import THREE, { CaptionTrack, CaptionOverlay } from '../threejs-shim.js';

const track   = CaptionTrack.parse(vttText);        // or .parseSrt(...)
const overlay = new CaptionOverlay(track, { fontSize: 34, anchor: 'bottom' });
overlay.setAutoScale(true);                          // style follows canvas height
renderer.setCaptions(overlay);

function frame(now) {
    renderer.captionTime = now / 1000;               // your playback clock
    renderer.render(scene, camera);                  // caption composited on top
    requestAnimationFrame(frame);
}
